import json
import sys
from pathlib import Path

from lupa import LuaRuntime

sys.path.insert(0, "araclar")
from lua_cikar import h

mod = Path("C:/Program Files (x86)/Steam/steamapps/workshop/content/108600")
mod /= "3803984183/mods/ProjectALifeNPCs/42.20/media/lua"
file = mod / "shared/ProjectALife/Missions/ALifeTaskBoardData.lua"

def read_json(path):
    return json.loads(Path(path).read_text(encoding="utf-8-sig"))

lua = LuaRuntime()
data = lua.execute(file.read_text(encoding="utf-8-sig"))
ui = read_json("teslim/IG_UI.json")
rd = read_json("teslim/RadioData.json")
l152 = {row["en"] for row in read_json("notlar/arsiv/2026-11-19/fable-dosyalari/L152_HAZIRLIK_540.json")["rows"]}
missing = {}
for task in data.tasks.values():
    for style, route in task.paths.items():
        for step in route.steps.values():
            en = step.label
            key = "IGUI_ALifeUI_" + h(en)
            if "{" not in en and en not in l152 and key not in ui:
                missing.setdefault(en, set()).add((task.id, style))

for en, contexts in sorted(missing.items()):
    radio_key = "RD_ALifeLine_" + h(en)
    print(en, sorted(contexts), radio_key if radio_key in rd else "")
